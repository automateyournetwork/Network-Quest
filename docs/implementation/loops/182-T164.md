The following check from the recipe checkout reconstructs all 442 client paths from the extracted package and compares each one against the earlier public release-candidate ledger.

```sh
python3 - .local/extracted-candidate <<'PY'
from pathlib import Path
import hashlib, json, sys
candidate = Path(sys.argv[1])
game = candidate / 'public/arcade/play/network-quest'
wrappers = {Path('.htaccess'), Path('api/v1/index.php'), Path('release.json')}
public = {p.relative_to(game): p for p in game.rglob('*') if p.is_file()}
assert wrappers <= public.keys()
files = {name: p for name, p in public.items() if name not in wrappers}
files[Path('.vite/manifest.json')] = candidate / 'metadata/vite-manifest.json'
pairs = [[name.as_posix(), hashlib.sha256(files[name].read_bytes()).hexdigest()] for name in sorted(files)]
reviewed = json.loads(Path('docs/evidence/spec001/refinements/production.json').read_text())
assert dict(pairs) == reviewed['distribution']
digest = hashlib.sha256(json.dumps(pairs, separators=(',', ':')).encode()).hexdigest()
assert digest == reviewed['sha256'] and len(pairs) == 442
print(len(pairs), sum(p.stat().st_size for p in files.values()), digest)
PY
```
